    set(set, parents(set))

proc showset(set) {
    foreach(set, per, n) {
        d(n) " " name(per) nl()
    }
    nl()
}

proc addtosetoverloadtest(per) {

    set(ancestorlist, ancestors(per)) /* create list of ancestors */
    valueof(ancestorlist) /* show the list to prove it is a list */
    set(set, personset()) /* create an empty personset */
    addtoset(set, ancestorlist) /* try out the new overload */
    call showset(set) /* prove that the new overload works */
}
